import json,sys
from PIL import Image,ImageDraw
items=[json.loads(x) for x in open('items.jsonl')]
start=int(sys.argv[1]); n=int(sys.argv[2]) if len(sys.argv)>2 else 5
rows=[]
for i,it in enumerate(items[start:start+n],start):
 w=Image.open(it['word_image']).convert('RGB'); l=Image.open(it['line_image']).convert('RGB')
 row=Image.new('RGB',(max(1400,w.width,l.width),w.height+l.height+40),'white'); d=ImageDraw.Draw(row); d.text((4,3),str(i+1)+' '+it['item_id'],fill='black'); row.paste(w,(0,25)); row.paste(l,(0,w.height+30)); rows.append(row)
out=Image.new('RGB',(max(r.width for r in rows),sum(r.height for r in rows)), 'white'); y=0
for r in rows: out.paste(r,(0,y)); y+=r.height
out.save('zoom/batch.png')
print('total',len(items),'batch',start+1,start+len(rows))
